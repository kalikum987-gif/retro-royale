<script>
const splashAudio = document.getElementById('splashAudio');
const clickAudio = document.getElementById('clickAudio');
const lobbyBgm = document.getElementById('lobbyBgm');

function playClick() {
clickAudio.currentTime = 0;
clickAudio.play().catch(() => {});
}

// Step 1: Splash से Login की तरफ बढ़ना
function startSequence() {
splashAudio.play().catch(() => {});
setTimeout(() => {
document.getElementById('splashScreen').classList.remove('active');
document.getElementById('loginScreen').classList.add('active');
}, 1800);
}

// Step 2: Login (Tap to Start) से Lobby में जाना
function goToLobby() {
playClick();
document.getElementById('loginScreen').classList.remove('active');
document.getElementById('lobbyScreen').classList.add('active');
lobbyBgm.volume = 0.5;
lobbyBgm.play().catch(() => {});
}
</script>
